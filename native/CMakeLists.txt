cmake_minimum_required(VERSION 3.15)
project(ZhIME LANGUAGES CXX)
add_library(ZhIME SHARED zhime_tsf.cpp exports.def)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/zhime_tsf.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/zhime_tsf.cpp" ZHIME_SOURCE_HASH)
string(SUBSTRING "${ZHIME_SOURCE_HASH}" 0 10 ZHIME_SOURCE_VERSION)
set_target_properties(ZhIME PROPERTIES OUTPUT_NAME "ZhIME-${ZHIME_SOURCE_VERSION}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-dll.txt" "ZhIME-${ZHIME_SOURCE_VERSION}.dll\n")
target_compile_features(ZhIME PRIVATE cxx_std_17)
target_compile_definitions(ZhIME PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ZhIME PRIVATE ole32 oleaut32 uuid advapi32 user32 gdi32 winhttp)
if(MSVC)
  target_compile_options(ZhIME PRIVATE /utf-8 /W4 /EHsc)
endif()
add_executable(rime_pool ../tools/rime_pool.cpp)
target_include_directories(rime_pool PRIVATE ../third_party/rime-api)
target_compile_features(rime_pool PRIVATE cxx_std_17)
target_compile_definitions(rime_pool PRIVATE UNICODE _UNICODE)
add_executable(tsf_probe probe.cpp)
target_compile_features(tsf_probe PRIVATE cxx_std_17)
target_compile_definitions(tsf_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(tsf_probe PRIVATE ole32 uuid)
add_executable(profile_tool profile_tool.cpp)
target_compile_features(profile_tool PRIVATE cxx_std_17)
target_compile_definitions(profile_tool PRIVATE UNICODE _UNICODE)
target_link_libraries(profile_tool PRIVATE ole32 uuid)
